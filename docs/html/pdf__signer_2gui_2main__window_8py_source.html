19class LoadPrivateKeyThread(QThread):

21 Thread for loading the private key.

34 def run(self):
35 """
36 Runs the private key loading thread.
37 """

49class SignPDFThread(QThread):
50 """
51 Thread for signing the PDF document.
52 """
53 finished = pyqtSignal(bool, str, str)
54
55 def __init__(self, private_key: bytes, pdf_path: str, signer_name: str):
56 """
57 Initializes the PDF signing thread.
58

68 def run(self):
69 """
70 Runs the PDF signing thread.
71 """
72 try:
73 signer = PDFSigner(self.private_key)

75 if success:
76 self.finished.emit(True, "Document signed successfully.", signed_pdf_path)

83class VerifyPDFThread(QThread):
84 """
85 Thread for verifying the PDF signature.
86 """
87 finished = pyqtSignal(bool, str, dict)
88
89 def __init__(self, public_key: bytes, pdf_path: str):
90 """
91 Initializes the PDF signature verification thread.
92
93 @param public_key RSA public key in PEM format.
94 @param pdf_path Path to the PDF document.
95 """
96 super().__init__()
97 self.public_key = public_key
98 self.pdf_path = pdf_path
99
100 def run(self):
101 """
102 Runs the PDF signature verification thread.
103 """
104 try:
105 verifier = PDFVerifier(self.public_key)
106 is_valid, message, signature_info = verifier.verify_pdf(self.pdf_path)
107 if signature_info is None:
108 signature_info = {}
109 self.finished.emit(is_valid, message, signature_info)
110 except Exception as e:
111 self.finished.emit(False, f"Error: {e}", {})

297 def sign_document(self):
298 """
299 Starts the document signing process.
300 """
301 if not self.drive_path or not self.encrypted_private_key:
302 QMessageBox.warning(self, "Error", "USB drive or private key not available.")
303 return
304
305 pin = self.pin_input.text()
306 if not pin:
307 QMessageBox.warning(self, "Error", "Please enter the PIN.")
308 return
309
310 try:
311 self.private_key = decrypt_private_key(self.encrypted_private_key, pin)
312 except Exception as e:
313 QMessageBox.warning(self, "Error", f"Failed to decrypt private key: {e}")
314 return
315
316 pdf_path = self.document_input.text()
317 if not os.path.isfile(pdf_path):
318 QMessageBox.warning(self, "Error", "Invalid PDF document path.")
319 return
320
321 signer_name = self.signer_name_input.text()
322 if not signer_name:
323 QMessageBox.warning(self, "Error", "Signer name is required.")
324 return
325
326
327 self.sign_progress_bar.setVisible(True)
328 self.sign_button.setEnabled(False)
329
330 self.sign_thread = SignPDFThread(self.private_key, pdf_path, signer_name)
331 self.sign_thread.finished.connect(self.on_sign_finished)

352 def verify_document(self):
353 """
354 Starts the signature verification process.
355 """
356 pdf_path = self.verify_document_input.text()
357 if not os.path.isfile(pdf_path):
358 QMessageBox.warning(self, "Error", "Invalid PDF document path.")
359 return
360
361 public_key_path = self.public_key_input.text()
362 if not os.path.isfile(public_key_path):
363 QMessageBox.warning(self, "Error", "Invalid public key path.")
364 return
365
366 try:
367 with open(public_key_path, "rb") as f:
368 self.public_key = f.read()
369 except Exception as e:
370 QMessageBox.warning(self, "Error", f"Failed to load public key: {e}")
371 return
372
373 self.verify_progress_bar.setVisible(True)
374 self.verify_button.setEnabled(False)
375
376 self.verify_thread = VerifyPDFThread(self.public_key, pdf_path)
377 self.verify_thread.finished.connect(self.on_verify_finished)
378 self.verify_thread.start()
