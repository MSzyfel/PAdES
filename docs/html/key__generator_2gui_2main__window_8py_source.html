24class KeyGenerationThread(QThread):
25 """
26 Thread for RSA key generation.
27 """
28 finished = pyqtSignal(bool, str)
29
30 def __init__(self, pin: str, drive_path: str, public_key_path: str):
31 """
32 Initializes the key generation thread.
33
34 @param pin: User PIN.
35 @param drive_path: Path to the USB drive.
36 @param public_key_path: Path to the public key file.
37 """
38 super().__init__()
39 self.pin = pin
40 self.drive_path = drive_path
41 self.public_key_path = public_key_path
42
43 def run(self):
44 """
45 Starts RSA key generation in a separate thread.
46 """
47 try:
48 generator = KeyGenerator()
49 generator.generate_keys()
50 generator.encrypt_private_key(self.pin)
51 private_saved, public_saved = generator.save_keys(
52 self.drive_path, self.public_key_path
53 )
54
55 if private_saved and public_saved:
56 self.finished.emit(True, "Keys were generated and saved successfully.")
57 elif private_saved:
58 self.finished.emit(False, "Private key saved, but saving the public key failed.")
59 elif public_saved:
60 self.finished.emit(False, "Public key saved, but saving the private key failed.")
61 else:
62 self.finished.emit(False, "Failed to save both keys.")
63 except Exception as e:
64 self.finished.emit(False, f"An error occurred: {e}")

169 def generate_keys(self):
170 """
171 Validates input and starts RSA key generation.
172 """
173 pin = self.pin_input.text()
174 confirm_pin = self.confirm_pin_input.text()
175
176 if len(pin) < 4:
177 QMessageBox.warning(self, "Error", "PIN must be at least 4 characters long.")
178 return
179 if pin != confirm_pin:
180 QMessageBox.warning(self, "Error", "PINs do not match.")
181 return
182
183 drive_path = self.drive_combo.currentText()
184 if drive_path == "No USB drives available":
185 QMessageBox.warning(self, "Error", "No USB drive selected.")
186 return
187
188 public_key_path = self.public_key_input.text()
189 if not public_key_path:
190 QMessageBox.warning(self, "Error", "Please select a public key file path.")
191 return
192
193 confirm = QMessageBox.question(
194 self, "Confirm",
195 "Are you sure you want to generate an RSA key pair?\n\n"
196 f"Private key will be saved to USB: {drive_path}\n"
197 f"Public key will be saved to: {public_key_path}",
198 QMessageBox.Yes | QMessageBox.No, QMessageBox.No
199 )
200 if confirm == QMessageBox.No:
201 return
202
203 self.progress_bar.setVisible(True)
204 self.generate_button.setEnabled(False)
205
206 self.thread = KeyGenerationThread(pin, drive_path, public_key_path)
207 self.thread.finished.connect(self.on_key_generation_finished)
208 self.thread.start()
