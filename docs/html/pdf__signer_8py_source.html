92 Adds a signature to the PDF document.

106 sig_dict = DictionaryObject({

108 NameObject("/Filter"): NameObject("/Adobe.PPKLite"),
109 NameObject("/SubFilter"): NameObject("/adbe.pkcs7.detached"),
110 NameObject("/Name"): createStringObject(signature.signer_name),
111 NameObject("/M"): createStringObject(signature.timestamp.strftime("D:%Y%m%d%H%M%S")),

113 NameObject("/ByteRange"): ArrayObject([NumberObject(0), NumberObject(0), NumberObject(0), NumberObject(0)]),
114 NameObject("/DocumentHash"): ByteStringObject(document_hash_base64),
115 NameObject("/DigestMethod"): NameObject(f"/{HASH_ALGORITHM.upper()}"),
116 NameObject("/DocumentHashEncoding"): NameObject("/BASE64"),
117 })
118
119 if "/AcroForm" not in writer._root_object:
120 writer._root_object[NameObject("/AcroForm")] = DictionaryObject({
121 NameObject("/SigFlags"): NumberObject(3),

123 })
