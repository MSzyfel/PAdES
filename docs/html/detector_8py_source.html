44 """
45 Returns a list of USB device paths on Windows.

47 try:
48 return [f"{l}:\\" for l in string.ascii_uppercase if
49 os.path.exists(f"{l}:\\") and ctypes.windll.kernel32.GetDriveTypeW(f"{l}:\\") == 2]

54 def _get_usb_drives_windows_wmi(self) -> List[str]:
55 """
56 Alternate method using WMI to detect USB drives on Windows.
57 """
58 drives = []
59 try:
60 import wmi
61 c = wmi.WMI()
62 for d in c.Win32_LogicalDisk():
63 if d.DriveType in [2, 3]:
64 p = f"{d.DeviceID}\\"
65 try:
66 os.listdir(p)
67 drives.append(p)
68 print(f"WMI detected: {p} (type: {d.DriveType})")
69 except:
70 print(f"No access: {p}")
71 except ImportError:
72 print("Install WMI: pip install WMI")
73 except Exception as e:
74 print(f"WMI error: {e}")

85 def _get_usb_drives_linux(self) -> List[str]:
86 """
87 Returns USB device paths on Linux.
88 """
89 base = "/media"
90 drives = []
91 if os.path.exists(base):
92 for user in os.listdir(base):
93 user_path = os.path.join(base, user)
94 if os.path.isdir(user_path):
95 for volume in os.listdir(user_path):
96 vp = os.path.join(user_path, volume)
97 if os.path.ismount(vp):
98 drives.append(vp)
99 return drives
